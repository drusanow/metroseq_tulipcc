# Tulip Studio: every app in modules/ that has an app.json, in one firmware.
#
# Adding an app = adding a folder modules/mod_<name>/ with its component
# (CMakeLists.txt, mod_<name>.h declaring `extern tmod_t mod_<name>;`) and an
# app.json (docs/APPS.md). Nothing here needs editing: the list below is
# found at build time and becomes apps.gen.c - the module table studio_run()
# gets, and the manifest the launcher reads out of studio.bin.
set(mod_root "${CMAKE_CURRENT_LIST_DIR}/../../../modules")

# The order is the Studio's: saved routes and patches go by position, so
# the first five keep theirs and new apps come after, by name.
set(known metroseq gintonix sloe wavetron drumster9)
file(GLOB jsons LIST_DIRECTORIES false "${mod_root}/mod_*/app.json")
list(SORT jsons)
set(found)
foreach(j ${jsons})
    get_filename_component(d "${j}" DIRECTORY)
    get_filename_component(d "${d}" NAME)
    string(REGEX REPLACE "^mod_" "" n "${d}")
    list(APPEND found ${n})
endforeach()
set(order)
foreach(n ${known})
    if(n IN_LIST found)
        list(APPEND order ${n})
    endif()
endforeach()
foreach(n ${found})
    if(NOT n IN_LIST order)
        list(APPEND order ${n})
    endif()
endforeach()
list(LENGTH order napps)
if(napps GREATER 16)
    message(FATAL_ERROR "Tulip Studio holds 16 apps at most (found ${napps})")
endif()

set(reqs)
foreach(n ${order})
    list(APPEND reqs mod_${n})
endforeach()

if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    set(inc "")
    set(tab "")
    set(man "")
    foreach(n ${order})
        file(READ "${mod_root}/mod_${n}/app.json" js)
        foreach(k name title role blurb)
            string(JSON v ERROR_VARIABLE err GET "${js}" ${k})
            if(err)
                message(FATAL_ERROR "modules/mod_${n}/app.json: no \"${k}\"")
            endif()
            string(REPLACE "\\" "\\\\" v "${v}")
            string(REPLACE "\"" "\\\"" v "${v}")
            set(${k} "${v}")
        endforeach()
        if(NOT name STREQUAL n)
            message(FATAL_ERROR "modules/mod_${n}/app.json: name must be \"${n}\"")
        endif()
        string(LENGTH "${name}" l1)
        string(LENGTH "${title}" l2)
        string(LENGTH "${blurb}" l3)
        if(l1 GREATER 15 OR l2 GREATER 15 OR l3 GREATER 79)
            message(FATAL_ERROR "modules/mod_${n}/app.json: name / title up to 15 characters, blurb 79")
        endif()
        if(role STREQUAL "sequencer")
            set(r APP_ROLE_SEQUENCER)
        elseif(role STREQUAL "drums")
            set(r APP_ROLE_DRUMS)
        elseif(role STREQUAL "synth")
            set(r APP_ROLE_SYNTH)
        else()
            message(FATAL_ERROR "modules/mod_${n}/app.json: role is sequencer, drums or synth")
        endif()
        string(APPEND inc "#include \"mod_${n}.h\"\n")
        string(APPEND tab "    &mod_${n},\n")
        string(APPEND man "        {\"${name}\", \"${title}\", ${r}, {0}, \"${blurb}\"},\n")
    endforeach()
    configure_file("${CMAKE_CURRENT_LIST_DIR}/apps.gen.c.in" "${CMAKE_CURRENT_BINARY_DIR}/apps.gen.c" @ONLY)
    set(gen "${CMAKE_CURRENT_BINARY_DIR}/apps.gen.c")
endif()

idf_component_register(SRCS "main.c" ${gen} REQUIRES ${reqs} tulip_studio tulip_hal)
# re-run when an app is added or its app.json changes
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${jsons} "${mod_root}")
